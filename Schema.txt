CREATE TABLE admins (
    admin_id     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    admin_name   VARCHAR2(100) NOT NULL,
    email        VARCHAR2(200) UNIQUE NOT NULL,
    password     VARCHAR2(100) NOT NULL,
    created_at   DATE DEFAULT SYSDATE
);

CREATE TABLE doctors (
    doctor_id      NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    doctor_name    VARCHAR2(100) NOT NULL,
    specialization VARCHAR2(100),
    age            NUMBER CHECK (age >= 18),
    gender         VARCHAR2(10) CHECK (gender IN ('Male', 'Female')),
    contact        VARCHAR2(30),
    email          VARCHAR2(200) UNIQUE,
    start_time     VARCHAR2(10),
    finish_time    VARCHAR2(10),
    fees           NUMBER(8,2),
    joining_date   DATE,
    password       VARCHAR2(100) NOT NULL
);
1 = Sun … 7 = Sat
CREATE TABLE doctor_availability (
    doctor_id   NUMBER NOT NULL,
    day_of_week NUMBER(1) NOT NULL,
    CONSTRAINT fk_da_doc
        FOREIGN KEY (doctor_id)
        REFERENCES doctors(doctor_id)
        ON DELETE CASCADE,
    CONSTRAINT uq_doc_day
        UNIQUE (doctor_id, day_of_week)
);
CREATE TABLE patients (
    patient_id    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    patient_name  VARCHAR2(100) NOT NULL,
    age           NUMBER,
    gender        VARCHAR2(10) CHECK (gender IN ('Male', 'Female')),
    phone         VARCHAR2(30),
    email         VARCHAR2(200) UNIQUE NOT NULL,
    password      VARCHAR2(100) NOT NULL,
    created_at    DATE DEFAULT SYSDATE
);
CREATE TABLE appointments (
    appointment_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    patient_id       NUMBER NOT NULL,
    doctor_id        NUMBER NOT NULL,
    appointment_date DATE NOT NULL,
    appointment_time VARCHAR2(5),
    status           VARCHAR2(20) DEFAULT 'BOOKED',
    created_at       DATE DEFAULT SYSDATE,
    CONSTRAINT fk_ap_patient
        FOREIGN KEY (patient_id)
        REFERENCES patients(patient_id),
    CONSTRAINT fk_ap_doctor
        FOREIGN KEY (doctor_id)
        REFERENCES doctors(doctor_id)
);

CREATE TABLE feedback (
    feedback_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    patient_id  NUMBER NOT NULL,
    message     VARCHAR2(2000) NOT NULL,
    created_at  DATE DEFAULT SYSDATE,
    CONSTRAINT fk_feedback_patient FOREIGN KEY (patient_id) REFERENCES patients(patient_id) ON DELETE CASCADE
);
